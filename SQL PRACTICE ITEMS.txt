CREATE DATABASE FoodDB;

USE FoodDB;

 

-- Items Table

CREATE TABLE Items (

    Item_ID INT PRIMARY KEY,

    Item_Name VARCHAR(50),

    Category VARCHAR(30),

    Price DECIMAL(10,2)

);

 

-- Food Orders Table

CREATE TABLE Food_Orders (

    Order_ID INT PRIMARY KEY,

    Item_ID INT,

    Customer_Name VARCHAR(50),

    Quantity INT,

    Order_Date DATE,

    FOREIGN KEY (Item_ID) REFERENCES Items(Item_ID)

);

SELECT 
    o.Order_ID, 
    o.Customer_Name, 
    i.Item_Name, 
    i.Price, 
    o.Quantity
FROM Food_Orders o
INNER JOIN Items i ON o.Item_ID = i.Item_ID;

SELECT DISTINCT o.Customer_Name
FROM Food_Orders o
INNER JOIN Items i ON o.Item_ID = i.Item_ID
WHERE i.Item_Name = 'Pizza';

SELECT * 
FROM Food_Orders
WHERE Quantity > 1;

SELECT DISTINCT o.Customer_Name
FROM Food_Orders o
INNER JOIN Items i ON o.Item_ID = i.Item_ID
WHERE i.Price > 150;

FROM Food_Orders o

FROM Food_Orders o
